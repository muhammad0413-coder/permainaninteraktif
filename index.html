Terminal Python Simulator Python 3.10 CLI

=== GAME MATEMATIKA PYTHON CLI ===

Pilih Tingkat Kesulitan:

1. Mudah (1-10)

2. Sedang (1-50)

3. Sulit (1-100)

Kode Python 3 Modular (Siap Push Ke GitHub): 

import random

def get_difficulty_range(choice):
    if choice == '1':
        return 1, 10
    elif choice == '2':
        return 1, 50
    elif choice == '3':
        return 1, 100
    return 1, 10

def generate_question(min_val, max_val):
    num1 = random.randint(min_val, max_val)
    num2 = random.randint(min_val, max_val)
    op = random.choice(['+', '-', '*'])
    if op == '+':
        ans = num1 + num2
    elif op == '-':
        ans = num1 - num2
    else:
        ans = num1 * num2
    return num1, op, num2, ans

def run_game():
    print("=== GAME MATEMATIKA CLI PYTHON ===")
    print("1. Mudah (1-10)\n2. Sedang (1-50)\n3. Sulit (1-100)")
    choice = input("Pilih tingkat kesulitan (1/2/3): ")
    min_v, max_v = get_difficulty_range(choice)
    
    score = 0
    total_q = 10
    
    for i in range(1, total_q + 1):
        n1, op, n2, correct_ans = generate_question(min_v, max_v)
        try:
            user_ans = int(input(f"Soal {i}/{total_q}: {n1} {op} {n2} = "))
            if user_ans == correct_ans:
                print(" Correct! (+10 pts)\n")
                score += 10
            else:
                print(f" Wrong! Jawaban benar: {correct_ans}\n")
        except ValueError:
            print(" Input tidak valid!\n")
            
    accuracy = (score / (total_q * 10)) * 100
    print("="*30)
    print(f"HASIL AKHIR - Total Skor: {score} / 100")
    print(f"Akurasi Jawaban: {accuracy:.1f}%")
    print("="*30)

if __name__ == "__main__":
    run_game()
